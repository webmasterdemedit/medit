// ============================================================ // RÉCUPÉRER LES LABELS (DISCIPLINES) // ============================================================ if (action === 'getLabels') { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Posts'); if (!sheet) { return reponse(false, 'Feuille "Posts" introuvable'); } var data = sheet.getDataRange().getValues(); var labels = []; for (var i = 1; i < data.length; i++) { var label = data[i][2]; // Colonne C if (label && labels.indexOf(label) === -1) { labels.push(label); } } return reponse(true, '', { labels: labels }); } // ============================================================ // RÉCUPÉRER LES NIVEAUX // ============================================================ if (action === 'getNiveaux') { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Niveaux'); if (!sheet) { return reponse(false, 'Feuille "Niveaux" introuvable'); } var data = sheet.getDataRange().getValues(); var niveaux = []; for (var i = 1; i < data.length; i++) { var niveau = data[i][0]; // Colonne A if (niveau) { niveaux.push(niveau); } } return reponse(true, '', { niveaux: niveaux }); } // ============================================================ // ENREGISTRER UNE DEMANDE D'INSCRIPTION // ============================================================ if (action === 'registerDemande') { var nom = e.parameter.nom; var contact = e.parameter.contact; var discipline = e.parameter.discipline; var niveau = e.parameter.niveau; var message = e.parameter.message || ''; var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Demandes'); if (!sheet) { return reponse(false, 'Feuille "Demandes" introuvable'); } sheet.appendRow([nom, contact, discipline, niveau, message, new Date()]); return reponse(true, 'Demande enregistrée'); }
