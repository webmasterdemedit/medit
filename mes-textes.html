function ouvrirAnnotations(articleId, titre) {
var reponse = toutesLesReponses[articleId] || null;
var annotations = reponse ? reponse.annotations : '';
var modal = document.createElement('div');
modal.className = 'modal-overlay';
modal.id = 'modalAnnotations';
modal.onclick = function(e) {
if (e.target === this) fermerAnnotations();
};
var html = '<div class="modal-contenu">';
html += '<h3>' + titre + '</h3>';
html += '<div class="modal-sous-titre">Annotations</div>';
if (annotations && annotations.trim() !== '') {
var parts = annotations.split('|');
for (var i = 0; i < parts.length; i++) {
var kv = parts[i].split(':');
if (kv.length === 2) {
var slideNum = kv[0].replace('slide', '');
var texte = kv[1];
html += '<div class="modal-annotation">';
html += '<strong>Slide ' + slideNum + '</strong>';
html += texte;
html += '</div>';
}
}
} else {
html += '<div class="aucune-annotation">Aucune annotation pour ce texte.</div>';
}
html += '<button class="modal-close" onclick="fermerAnnotations()">Fermer</button>';
html += '</div>';
modal.innerHTML = html;
document.body.appendChild(modal);
}
function fermerAnnotations() {
var modal = document.getElementById('modalAnnotations');
if (modal) modal.remove();
}

function estPostLu(post) {
var reponse = toutesLesReponses[post.id];
if (!reponse) return false;
var bonnes = reponse.bonnes || 0;
var total = reponse.total || 0;
if (total === 1 && bonnes === 1) return true;
if (total > 0 && bonnes >= Math.ceil(total / 2)) return true;
return false;
}

function afficherOnglet(onglet, categoriesTriees, categories) {
var container = document.getElementById('contenu-onglet');
if (!container) return;
var html = '';
html += `<div class="sous-tabs-container" id="sous-tabs">`;
var totalPosts = 0;
categoriesTriees.forEach(function(cat) {
categories[cat].forEach(function(post) {
if (onglet === 'a-lire' && !estPostLu(post)) totalPosts++;
if (onglet === 'lus' && estPostLu(post)) totalPosts++;
});
});
html += `
<button class="sous-tab-btn active" data-sous-tab="toutes" onclick="switchCategorie('toutes')">
Toutes
<span class="compteur-sous-tab">${totalPosts}</span>
</button>
`;
categoriesTriees.forEach(function(cat) {
var count = 0;
categories[cat].forEach(function(post) {
if (onglet === 'a-lire' && !estPostLu(post)) count++;
if (onglet === 'lus' && estPostLu(post)) count++;
});
if (count > 0) {
html += `
<button class="sous-tab-btn" data-sous-tab="${cat}" onclick="switchCategorie('${cat}')">
${cat}
<span class="compteur-sous-tab">${count}</span>
</button>

}
});
html += `</div>`;
html += `<div id="sous-contenu"></div>`;
container.innerHTML = html;
var postsFiltres = filtrerPosts(onglet, 'toutes');
afficherPosts(postsFiltres);
}
function switchOnglet(onglet) {
ongletActuel = onglet;
document.querySelectorAll('.tab-btn').forEach(function(el) {

categories[cat].push(post);

afficherOnglet(onglet, categoriesTriees, categories);
}
function switchCategorie(categorie) {
var container = document.getElementById('sous-tabs');
if (container) {
container.querySelectorAll('.sous-tab-btn').forEach(function(el) {

el.classList.add('active');
}
});
}
var postsFiltres = filtrerPosts(ongletActuel, categorie);
afficherPosts(postsFiltres);
}
function filtrerPosts(onglet, categorie) {
var resultats = [];
postsData.forEach(function(post) {
var estLu = estPostLu(post);
if (onglet === 'a-lire' && estLu) return;
if (onglet === 'lus' && !estLu) return;
var catPost = post.categorie || 'Sans catégorie';
if (categorie !== 'toutes' && catPost !== categorie) return;
resultats.push(post);
});
return resultats;
}

var html = '<div class="grille">';
posts.forEach(function(post) {

var retenir = '';
