// PARSER - NETTOYAGE DE TOUS LES |
// ============================================================
function parserContenu(texte) {
var result = { slides: [], quizs: [], openQuestions: [], retenir: [] };
if (!texte) return result;
// ===== NETTOYAGE AGRESSIF DES | =====
// Supprimer tous les | isolés (avec ou sans espaces)
texte = texte.replace(/\s*\|\s*/g, ' ');
// Supprimer les | en fin de ligne
texte = texte.replace(/\|\s*$/gm, '');
// Supprimer les | en début de ligne
texte = texte.replace(/^\s*\|\s*/gm, '');
// Supprimer les doubles espaces
texte = texte.replace(/\s{2,}/g, ' ');
var patterns = ['-s-', '-q-', '-v-', '-f-', '-qs-', '-r-', '-i-'];
var indices = [];
for (var p = 0; p < patterns.length; p++) {
var pos = texte.indexOf(patterns[p]);
while (pos !== -1) {
indices.push({ pos: pos, pattern: patterns[p] });
pos = texte.indexOf(patterns[p], pos + 1);
}
}
indices.sort(function(a, b) { return a.pos - b.pos; });
var elements = [];
for (var i = 0; i < indices.length; i++) {
var start = indices[i].pos;
var end = (i + 1 < indices.length) ? indices[i + 1].pos : texte.length;
var element = texte.substring(start, end).trim();
if (element) elements.push(element);
}
var quizCourant = null;
for (var i = 0; i < elements.length; i++) {
var ligne = elements[i].trim();
if (!ligne) continue;
if (ligne.startsWith('-s-')) {
result.slides.push({ type: 'slide', contenu: ligne.replace('-s-', '').trim() || 'Slide vide' });
continue;
}
if (ligne.startsWith('-q-')) {
if (quizCourant && quizCourant.reponses.length > 0) {
result.quizs.push(quizCourant);
}
quizCourant = { question: ligne.replace('-q-', '').trim() || 'Question', reponses: [] };
continue;
}
if (ligne.startsWith('-v-')) {
if (quizCourant) {
quizCourant.reponses.push({ texte: ligne.replace('-v-', '').trim() || 'Vrai', correct: true });
}
continue;
}
if (ligne.startsWith('-f-')) {
if (quizCourant) {
quizCourant.reponses.push({ texte: ligne.replace('-f-', '').trim() || 'Faux', correct: false });
}
continue;
}
if (ligne.startsWith('-qs-')) {
result.openQuestions.push(ligne.replace('-qs-', '').trim() || 'Question ouverte');
continue;
}
if (ligne.startsWith('-r-')) {
var retenirText = ligne.replace('-r-', '').trim();
if (retenirText) {
var mots = retenirText.split(',').map(function(m) { return m.trim(); });
result.retenir = mots.filter(function(m) { return m.length > 0; });
}
continue;
}
if (ligne.startsWith('-i-')) {
continue;
}
}
if (quizCourant && quizCourant.reponses.length > 0) {
result.quizs.push(quizCourant);
}
return result;
}

function preparerSlides(post) {

var contenu = post.contenu || '';
var parsed = parserContenu(contenu);

});

});
parsed.openQuestions.forEach(function(q) {
slides.push({ type: 'open', question: q, index: slides.length });
});
if (parsed.retenir && parsed.retenir.length > 0) {
slides.push({ type: 'retenir', mots: parsed.retenir });
}
etat.slides = slides;

function afficherFeedbackQuiz(quizIndex) {
var slide = etat.slides.find(function(s) { return s.type === 'quiz' && s.index === quizIndex; });
if (!slide) return;
var reponseData = etat.reponsesQuiz[quizIndex];
if (!reponseData) return;
var slideEl = document.getElementById('slide-' + etat.slides.indexOf(slide));
if (!slideEl) return;
var reponses = slideEl.querySelectorAll('.quiz-reponse');
var feedback = document.getElementById('quizFeedback-' + quizIndex);
var bonneReponse = document.getElementById('quizBonneReponse-' + quizIndex);
var btnContinuer = document.getElementById('btnContinuer-' + quizIndex);
reponses.forEach(function(el, idx) {
el.classList.remove('correcte', 'incorrecte');
var icon = el.querySelector('.result-icon');
if (idx === reponseData.reponseIndex) {
if (reponseData.correct) { el.classList.add('correcte'); icon.textContent = '✓'; }
else { el.classList.add('incorrecte'); icon.textContent = '✗'; }
} else if (slide.reponses[idx] && slide.reponses[idx].correct) {
el.classList.add('correcte');
icon.textContent = '✓';
} else {
icon.textContent = '';
}
});
if (reponseData.correct) {
bonneReponse.textContent = '✓ Bonne réponse !';
bonneReponse.className = 'bonne-reponse';
} else {
var bonne = slide.reponses.find(function(r) { return r.correct; });
bonneReponse.textContent = '✗ La bonne réponse était : "' + (bonne ? bonne.texte : '—') + '"';
bonneReponse.className = 'mauvaise-reponse';
}
feedback.classList.add('visible');
setTimeout(function() {
btnContinuer.classList.add('visible');
}, 400);
verifierQuizComplets();
}
function verifierQuizComplets() {
var quizSlides = etat.slides.filter(function(s) { return s.type === 'quiz'; });
var tousRepondus = quizSlides.every(function(q) {
return etat.reponsesQuiz[q.index] !== undefined;
});
if (!tousRepondus) return;
var totalBonnes = 0, totalQuestions = 0;
quizSlides.forEach(function(q) {
var rep = etat.reponsesQuiz[q.index];
if (rep) { totalQuestions++; if (rep.correct) totalBonnes++; }
});
etat.totalBonnes = totalBonnes;
etat.totalQuestions = totalQuestions;
var seuil = Math.ceil(totalQuestions / 2);
etat.statutFinal = (totalQuestions > 0 && totalBonnes >= seuil) ? 'valide' : 'non-valide';
if (totalQuestions === 0) etat.statutFinal = 'en-attente';
mettreAJourBadge();
sauvegarderQuiz();
}
