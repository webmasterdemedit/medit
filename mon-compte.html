// ============================================================ // RÉCUPÉRER L'HISTORIQUE D'UN ÉTUDIANT (feuille "Reponses") // ============================================================ if (action === 'getHistorique') { var nom = e.parameter.nom; var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Reponses'); if (!sheet) { return reponse(false, 'Feuille "Reponses" introuvable'); } var data = sheet.getDataRange().getValues(); var historique = []; for (var i = 1; i < data.length; i++) { if (data[i][1] === nom) { var bonnes = parseInt(data[i][5]) || 0; var total = parseInt(data[i][6]) || 0; var valide = (total > 0 && bonnes >= Math.ceil(total / 2)); var enCours = (data[i][7] && data[i][7].trim().length > 0); // réponse ouverte remplie historique.push({ date: data[i][0] || '', temps: data[i][8] || '', valide: valide, enCours: enCours && !valide }); } } // Trier par date (du plus récent au plus ancien) historique.sort(function(a, b) { if (a.date && b.date) { return new Date(b.date) - new Date(a.date); } return 0; }); return reponse(true, '', { historique: historique }); }
