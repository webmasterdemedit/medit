function preparerPages(post) { var texteComplet = reconstituerTexte(post.contenu || ''); if (!texteComplet) { etat.pages = ['Ce texte est vide.']; return; } var mots = texteComplet.split(' '); var pages = []; // Créer un conteneur de mesure var mesure = document.createElement('div'); mesure.style.cssText = 'position:fixed;top:-9999px;left:0;width:' + Math.min(window.innerWidth - 40, 700) + 'px;padding:40px 20px;box-sizing:border-box;font-family:Georgia,serif;font-size:22px;line-height:2.2;text-align:justify;'; document.body.appendChild(mesure); var hauteurDispo = window.innerHeight - 80; var pageActuelle = []; var hauteurActuelle = 0; for (var i = 0; i < mots.length; i++) { pageActuelle.push(mots[i]); mesure.innerHTML = pageActuelle.join(' '); hauteurActuelle = mesure.scrollHeight; if (hauteurActuelle > hauteurDispo) { // Retirer le dernier mot pageActuelle.pop(); pages.push(pageActuelle.join(' ')); pageActuelle = [mots[i]]; } } if (pageActuelle.length > 0) { pages.push(pageActuelle.join(' ')); } document.body.removeChild(mesure); if (pages.length === 0) pages = ['Ce texte est vide.']; etat.pages = pages; }
